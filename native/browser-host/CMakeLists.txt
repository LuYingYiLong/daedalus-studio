cmake_minimum_required(VERSION 3.24)
project(daedalus_browser_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(channel development stable)
  add_executable(daedalus-browser-${channel} main.cpp)
  target_compile_definitions(daedalus-browser-${channel} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX BROWSER_CHANNEL=L"${channel}")
  target_compile_options(daedalus-browser-${channel} PRIVATE /W4 /EHsc /utf-8)
  target_link_libraries(daedalus-browser-${channel} PRIVATE advapi32 windowsapp)
endforeach()
